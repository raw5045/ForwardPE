# Forward P/E Dashboard MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first internal-only Forward P/E dashboard using FMP data, Neon Postgres snapshots, NTM EPS calculations, aggregate index/ETF valuation, data-health metrics, and private access controls.

**Architecture:** Create a Next.js App Router application with a small server-side data layer. Keep FMP-specific code behind provider interfaces, store normalized snapshots in Neon through Drizzle, calculate valuation metrics in pure TypeScript modules, and render server-side dashboard pages from query helpers.

**Tech Stack:** Next.js, TypeScript, React, Drizzle ORM, Neon Postgres, Vitest, Zod, Recharts, Lucide React, plain CSS.

---

## Scope

This plan implements the private/internal MVP described in `docs/superpowers/specs/2026-05-06-forward-pe-dashboard-design.md`.

The MVP includes:

- S&P 500 index view.
- Nasdaq-100 / NDX view.
- QQQ view.
- Sector ETF views for XLK, XLF, XLV, XLY, XLC, XLI, XLP, XLE, XLU, XLB, XLRE.
- S&P 500 constituent screener and stock detail pages.
- NTM EPS calculation using the quarterly-sum primary method and fiscal-year interpolation fallback.
- Count-based and weight-based method coverage metrics.
- FMP provider adapters.
- Neon/Drizzle schema.
- Manual ingestion command and admin trigger.
- Internal auth gate.

The MVP does not implement a public-safe valuation model, billing, public API access, or historical point-in-time backfill beyond FMP data available to the private account.

- [ ] **Step 4: Implement NTM calculation**

Create `src/lib/valuation/ntm.ts` with:

```ts
import type { EstimateInput, StockValuationInput, StockValuationResult } from "./types";

function periodKey(estimate: EstimateInput) {
  return estimate.periodType === "quarter"
    ? `${estimate.fiscalYear}Q${estimate.fiscalQuarter}`
    : `${estimate.fiscalYear}`;
}

function averageAnalystCount(estimates: EstimateInput[]) {
  const counts = estimates
    .map((estimate) => estimate.analystCount)
    .filter((count): count is number => typeof count === "number");

  if (counts.length === 0) {
    return null;
  }

  return Math.round(counts.reduce((sum, count) => sum + count, 0) / counts.length);
}

function monthDifferenceInclusive(startDate: Date, endDate: Date) {
  const yearMonths = (endDate.getUTCFullYear() - startDate.getUTCFullYear()) * 12;
  const monthDelta = endDate.getUTCMonth() - startDate.getUTCMonth();
  return Math.max(0, Math.min(12, yearMonths + monthDelta + 1));
}

function unavailable(input: StockValuationInput, reason: StockValuationResult["unavailableReason"], fallbackReason?: "missing_quarterly_estimates"): StockValuationResult {
  return {
    symbol: input.symbol,
    method: "unavailable",
    price: input.price,
    ntmEps: null,
    earningsYield: null,
    forwardPe: null,
    estimatePeriods: [],
    analystCount: averageAnalystCount(input.estimates),
    fallbackReason,
    unavailableReason: reason
  };
}

export function calculateStockValuation(input: StockValuationInput): StockValuationResult {
  if (input.price === null || input.price <= 0) {
    return unavailable(input, "missing_price");
  }

  const quarterlyEstimates = input.estimates
    .filter((estimate) => estimate.periodType === "quarter" && !estimate.reported && typeof estimate.epsAvg === "number")
    .sort((a, b) => a.periodEndDate.localeCompare(b.periodEndDate))
    .slice(0, 4);

  if (quarterlyEstimates.length === 4) {
    const ntmEps = quarterlyEstimates.reduce((sum, estimate) => sum + Number(estimate.epsAvg), 0);

    if (ntmEps <= 0) {
      return unavailable(input, "non_positive_ntm_eps");
    }

    return {
      symbol: input.symbol,
      method: "quarterly_sum",
      price: input.price,
      ntmEps,
      earningsYield: ntmEps / input.price,
      forwardPe: input.price / ntmEps,
      estimatePeriods: quarterlyEstimates.map(periodKey),
      analystCount: averageAnalystCount(quarterlyEstimates)
    };
  }

  const annualEstimates = input.estimates
    .filter((estimate) => estimate.periodType === "annual" && !estimate.reported && typeof estimate.epsAvg === "number")
    .sort((a, b) => a.periodEndDate.localeCompare(b.periodEndDate))
    .slice(0, 2);

  if (annualEstimates.length < 2) {
    return unavailable(input, "missing_annual_estimates", "missing_quarterly_estimates");
  }

  const valuationDate = new Date(`${input.valuationDate}T00:00:00.000Z`);
  const fy1EndDate = new Date(`${annualEstimates[0].periodEndDate}T00:00:00.000Z`);
  const remainingCurrentFyMonths = monthDifferenceInclusive(valuationDate, fy1EndDate);
  const remainingCurrentFyWeight = remainingCurrentFyMonths / 12;
  const nextFyWeight = 1 - remainingCurrentFyWeight;
  const ntmEps = Number(annualEstimates[0].epsAvg) * remainingCurrentFyWeight + Number(annualEstimates[1].epsAvg) * nextFyWeight;

  if (ntmEps <= 0) {
    return unavailable(input, "non_positive_ntm_eps", "missing_quarterly_estimates");
  }

  return {
    symbol: input.symbol,
    method: "fiscal_year_interpolation",
    price: input.price,
    ntmEps,
    earningsYield: ntmEps / input.price,
    forwardPe: input.price / ntmEps,
    estimatePeriods: annualEstimates.map(periodKey),
    analystCount: averageAnalystCount(annualEstimates),
    fallbackReason: "missing_quarterly_estimates"
  };
}
```

import type { AggregateValuationInput, AggregateValuationResult } from "./types";

export function calculateAggregateValuation(input: AggregateValuationInput): AggregateValuationResult {
  const totals = input.constituents.reduce(
    (acc, constituent) => {
      acc.constituentCount += 1;

      const hasUsableValue =
        constituent.price !== null &&
        constituent.price > 0 &&
        constituent.ntmEps !== null &&
        constituent.ntmEps > 0 &&
        constituent.method !== "unavailable";

      if (constituent.method === "quarterly_sum") {
        acc.quarterlySumCount += 1;
        acc.quarterlySumWeight += constituent.weight;
      } else if (constituent.method === "fiscal_year_interpolation") {
        acc.fiscalYearInterpolationCount += 1;
        acc.fiscalYearInterpolationWeight += constituent.weight;
      } else {
        acc.unavailableCount += 1;
        acc.unavailableWeight += constituent.weight;
      }

      if (hasUsableValue) {
        acc.coveredConstituentCount += 1;
        acc.coveredWeight += constituent.weight;
        acc.earningsYield += constituent.weight * (constituent.ntmEps / constituent.price);
      } else {
        acc.missingWeight += constituent.weight;
        if (constituent.method !== "unavailable") {
          acc.unavailableWeight += constituent.weight;
          acc.unavailableCount += 1;
        }
      }

      return acc;
    },
    {
      earningsYield: 0,
      coveredWeight: 0,
      missingWeight: 0,
      quarterlySumWeight: 0,
      fiscalYearInterpolationWeight: 0,
      unavailableWeight: 0,
      constituentCount: 0,
      coveredConstituentCount: 0,
      quarterlySumCount: 0,
      fiscalYearInterpolationCount: 0,
      unavailableCount: 0
    }
  );

  return {
    symbol: input.symbol,
    valuationDate: input.valuationDate,
    method: "aggregate",
    forwardPe: totals.earningsYield > 0 ? 1 / totals.earningsYield : null,
    earningsYield: totals.earningsYield > 0 ? totals.earningsYield : null,
    ...totals
  };
}
```

import type { ProviderConstituent, ProviderEstimate, ProviderHolding, ProviderQuote } from "../types";

function toNumber(value: unknown): number | null {
  if (typeof value === "number" && Number.isFinite(value)) {
    return value;
  }

  if (typeof value === "string" && value.trim() !== "") {
    const parsed = Number(value);
    return Number.isFinite(parsed) ? parsed : null;
  }

  return null;
}

function parseFiscalQuarter(periodEndDate: string) {
  const month = Number(periodEndDate.slice(5, 7));
  if ([1, 2, 3].includes(month)) return 1;
  if ([4, 5, 6].includes(month)) return 2;
  if ([7, 8, 9].includes(month)) return 3;
  return 4;
}

export function mapFmpEstimate(row: Record<string, unknown>, periodType: "annual" | "quarter"): ProviderEstimate {
  const periodEndDate = String(row.date ?? row.periodEndDate);

  return {
    symbol: String(row.symbol),
    periodType,
    fiscalYear: Number(periodEndDate.slice(0, 4)),
    fiscalQuarter: periodType === "quarter" ? parseFiscalQuarter(periodEndDate) : null,
    periodEndDate,
    epsAvg: toNumber(row.estimatedEpsAvg),
    epsLow: toNumber(row.estimatedEpsLow),
    epsHigh: toNumber(row.estimatedEpsHigh),
    analystCount: toNumber(row.numberAnalystEstimatedEps),
    raw: row
  };
}

export function mapFmpQuote(row: Record<string, unknown>): ProviderQuote {
  const price = toNumber(row.price);

  if (price === null) {
    throw new Error(`FMP quote for ${String(row.symbol)} is missing price`);
  }

  return {
    symbol: String(row.symbol),
    price,
    raw: row
  };
}

export function mapFmpSp500Constituent(row: Record<string, unknown>): ProviderConstituent {
  return {
    symbol: String(row.symbol),
    name: String(row.name),
    sector: typeof row.sector === "string" ? row.sector : null,
    raw: row
  };
}

export function mapFmpHolding(row: Record<string, unknown>): ProviderHolding {
  const symbol = String(row.asset ?? row.symbol);
  const weightValue = toNumber(row.weightPercentage ?? row.weight);

  return {
    symbol,
    name: typeof row.name === "string" ? row.name : null,
    weight: weightValue === null ? 0 : weightValue / 100,
    raw: row
  };
}

export class FmpClient {
  private readonly baseUrl = "https://financialmodelingprep.com/stable";

  constructor(private readonly apiKey = process.env.FMP_API_KEY) {
    if (!apiKey) {
      throw new Error("FMP_API_KEY is required");
    }
  }

  async get<T>(path: string, params: Record<string, string | number | undefined> = {}): Promise<T> {
    const url = new URL(`${this.baseUrl}${path}`);
    url.searchParams.set("apikey", this.apiKey);

    for (const [key, value] of Object.entries(params)) {
      if (value !== undefined) {
        url.searchParams.set(key, String(value));
      }
    }

    const response = await fetch(url);

    if (!response.ok) {
      throw new Error(`FMP request failed ${response.status}: ${url.pathname}`);
    }

    return response.json() as Promise<T>;
  }
}

import { calculateAggregateValuation } from "@/lib/valuation/aggregate";
import { calculateStockValuation } from "@/lib/valuation/ntm";
import type { EstimateInput } from "@/lib/valuation/types";
import { sectorEtfs } from "@/lib/universe/defaults";
import type { IngestionProvider, IngestionRepository, IngestionResult } from "./types";

type RunDailyIngestionInput = {
  repository: IngestionRepository;
  provider: IngestionProvider;
  runDate: string;
};

function providerEstimateToInput(estimate: {
  periodType: "quarter" | "annual";
  fiscalYear: number;
  fiscalQuarter: number | null;
  periodEndDate: string;
  epsAvg: number | null;
  analystCount: number | null;
}, valuationDate: string): EstimateInput {
  return {
    periodType: estimate.periodType,
    fiscalYear: estimate.fiscalYear,
    fiscalQuarter: estimate.fiscalQuarter ?? undefined,
    periodEndDate: estimate.periodEndDate,
    epsAvg: estimate.epsAvg,
    analystCount: estimate.analystCount,
    reported: estimate.periodEndDate < valuationDate
  };
}

export async function runDailyIngestion(input: RunDailyIngestionInput): Promise<IngestionResult> {
  const runId = await input.repository.startIngestionRun({ runDate: input.runDate, kind: "daily" });
  const errors: string[] = [];
  let symbolsProcessed = 0;

  try {
    const sp500Constituents = await input.provider.getSp500Constituents();

    for (const constituent of sp500Constituents) {
      await input.repository.upsertInstrument({
        symbol: constituent.symbol,
        name: constituent.name,
        type: "stock",
        sector: constituent.sector,
        active: true
      });
      await input.repository.upsertGroupMembership({
        groupSlug: "sp500",
        symbol: constituent.symbol,
        effectiveDate: input.runDate,
        source: "fmp",
        raw: constituent.raw
      });
    }

    const symbols = sp500Constituents.map((constituent) => constituent.symbol);
    const quotes = await input.provider.getQuotes([...symbols, "QQQ", ...sectorEtfs]);
    const quoteBySymbol = new Map(quotes.map((quote) => [quote.symbol, quote]));

    for (const quote of quotes) {
      await input.repository.upsertPriceSnapshot({
        symbol: quote.symbol,
        snapshotDate: input.runDate,
        price: quote.price,
        source: "fmp",
        raw: quote.raw
      });
    }

    for (const symbol of symbols) {
      try {
        const quarterly = await input.provider.getEstimates(symbol, "quarter");
        const annual = await input.provider.getEstimates(symbol, "annual");

        for (const estimate of [...quarterly, ...annual]) {
          await input.repository.upsertEstimateSnapshot({
            symbol,
            snapshotDate: input.runDate,
            estimate,
            source: "fmp"
          });
        }

        const valuation = calculateStockValuation({
          symbol,
          valuationDate: input.runDate,
          price: quoteBySymbol.get(symbol)?.price ?? null,
          fiscalYearEndMonth: 12,
          estimates: [...quarterly, ...annual].map((estimate) => providerEstimateToInput(estimate, input.runDate))
        });

        await input.repository.upsertValuationSnapshot({
          symbol,
          snapshotDate: input.runDate,
          valuation,
          source: "fmp_consensus_ntm_private"
        });
        symbolsProcessed += 1;
      } catch (error) {
        errors.push(`${symbol}: ${error instanceof Error ? error.message : String(error)}`);
      }
    }

    const latestConstituents = await input.repository.getLatestGroupConstituents("sp500");
    const stockValuations = await input.repository.getLatestStockValuations(input.runDate, latestConstituents.map((row) => row.symbol));
    const valuationBySymbol = new Map(stockValuations.map((row) => [row.symbol, row]));
    const aggregate = calculateAggregateValuation({
      symbol: "SP500",
      valuationDate: input.runDate,
      constituents: latestConstituents.map((row) => ({
        symbol: row.symbol,
        weight: row.weight,
        price: valuationBySymbol.get(row.symbol)?.price ?? null,
        ntmEps: valuationBySymbol.get(row.symbol)?.ntmEps ?? null,
        method: valuationBySymbol.get(row.symbol)?.method ?? "unavailable"
      }))
    });

    await input.repository.upsertValuationSnapshot({
      symbol: "SP500",
      snapshotDate: input.runDate,
      valuation: aggregate,
      source: "fmp_consensus_ntm_private"
    });

    const status = errors.length === 0 ? "succeeded" : "partial";
    await input.repository.finishIngestionRun(runId, { status, symbolsProcessed, errors });
    return { status, runId, runDate: input.runDate, symbolsProcessed, errors };
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    await input.repository.failIngestionRun(runId, message);
    return { status: "failed", runId, runDate: input.runDate, symbolsProcessed, errors: [message] };
  }
}
```

import { NextResponse, type NextRequest } from "next/server";

export function isAuthorizedRequest(request: Request, token = process.env.INTERNAL_ACCESS_TOKEN) {
  if (!token) {
    return true;
  }

  return request.headers.get("authorization") === `Bearer ${token}`;
}

export function middleware(request: NextRequest) {
  if (isAuthorizedRequest(request)) {
    return NextResponse.next();
  }

  return new NextResponse("Unauthorized", {
    status: 401,
    headers: {
      "www-authenticate": "Bearer"
    }
  });
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"]
};

export async function POST(request: Request) {
